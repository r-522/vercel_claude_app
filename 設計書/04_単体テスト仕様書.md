## 2. 前提条件・注意事項

> **重要**: 現在の `package.json` には `test` スクリプトが定義されていません。  
> 以下のテストケースを実行するには、別途テスト環境の構築が必要です。

### 2.1 テストフレームワーク（導入手順）

```bash
npm install --save-dev jest @types/jest ts-jest
npm install --save-dev @testing-library/react @testing-library/user-event @testing-library/jest-dom
npm install --save-dev jest-environment-jsdom
```

`package.json` に以下のスクリプトを追加：

```json
{
  "scripts": {
    "test": "jest",
    "test:coverage": "jest --coverage",
    "test:watch": "jest --watch"
  }
}
```

```typescript
describe('POST /api/auth/verify', () => {
  beforeEach(() => {
    process.env.ACCESS_CODE = '1234'
    process.env.COOKIE_SECRET = 'test-secret-that-is-at-least-32chars!!'
    // rate-limiter の store をリセット（モックまたは間接呼び出し）

    const req = new Request('http://localhost/api/auth/verify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-forwarded-for': '1.2.3.4' },
      body: JSON.stringify({ code: '1234' })
    })
    const res = await POST(req as NextRequest)

    expect(res.status).toBe(200)
    const body = await res.json()
    expect(body.success).toBe(true)
    expect(res.headers.get('Set-Cookie')).toContain('auth_session=')
    expect(res.headers.get('Set-Cookie')).toContain('HttpOnly')
  })
})
```

**入力**: `{ code: '1234' }`（`ACCESS_CODE=1234`）  
**期待結果**: 200、`Set-Cookie: auth_session=<JWT>...`、`{ success: true }`

### 6.10 useImageAttachments フック

```typescript
describe('useImageAttachments', () => {
  test('add で画像が追加される', () => {
    const file = new File(['dummy'], 'test.png', { type: 'image/png' })
    const { result } = renderHook(() => useImageAttachments())
    act(() => { result.current.add([file]) })
    expect(result.current.attached).toHaveLength(1)
    expect(result.current.attached[0].file).toBe(file)
  })

  test('image/* 以外のファイルは無視される', () => {
    const file = new File(['dummy'], 'test.pdf', { type: 'application/pdf' })
    const { result } = renderHook(() => useImageAttachments())
    act(() => { result.current.add([file]) })
    expect(result.current.attached).toHaveLength(0)
  })

  test('remove で特定の画像が削除される', () => {
    const file1 = new File(['a'], 'a.png', { type: 'image/png' })
    const file2 = new File(['b'], 'b.png', { type: 'image/png' })
    const { result } = renderHook(() => useImageAttachments())
    act(() => { result.current.add([file1, file2]) })
    act(() => { result.current.remove(0) })
    expect(result.current.attached).toHaveLength(1)
    expect(result.current.attached[0].file).toBe(file2)
  })

  test('clear で全て削除される', () => {
    const file = new File(['a'], 'a.png', { type: 'image/png' })
    const { result } = renderHook(() => useImageAttachments())
    act(() => { result.current.add([file]) })
    act(() => { result.current.clear() })
    expect(result.current.attached).toHaveLength(0)
  })
})
```
