## 4. API エンドポイント詳細設計

### 4.1 POST /api/auth/verify

**ファイル**: `src/app/api/auth/verify/route.ts`

```typescript
const GENERIC_ERROR = 'コードが正しくありません'

export async function POST(request: NextRequest): Promise<Response> {
  // 1. IP 解決（x-forwarded-for の第1セグメント or '127.0.0.1'）
  const ip = (request.headers.get('x-forwarded-for')
    ?.split(',')[0].trim()) ?? '127.0.0.1'

  // 2. レート制限チェック（コード比較より前 → タイミング攻撃対策）
  const { allowed } = checkRateLimit(ip)
  if (!allowed) {
    return Response.json(GENERIC_ERROR, { status: 429 })
  }

  // 3. リクエストボディ解析
  let body: { code?: unknown }
  try {
    body = await request.json()
  } catch {
    return Response.json(GENERIC_ERROR, { status: 400 })
  }

  // 4. ACCESS_CODE 環境変数取得
  const accessCode = process.env.ACCESS_CODE
  if (!accessCode) {
    console.error('ACCESS_CODE is not configured')
    return Response.json(GENERIC_ERROR, { status: 500 })
  }

  // 5. コード検証（平文比較。定数時間比較ではない）
  const { code } = body
  if (typeof code !== 'string' || code !== accessCode) {
    return Response.json(GENERIC_ERROR, { status: 401 })
  }

  // 6. JWT 生成・Set-Cookie レスポンス
  const token = await signAuthCookie()
  return Response.json(
    { success: true },
    { status: 200, headers: { 'Set-Cookie': buildCookieHeader(token) } }
  )
}
```

**入出力仕様**:

| 項目 | 詳細 |
|------|------|
| リクエスト | `{ code: string }` |
| 成功レスポンス | 200 + `Set-Cookie: auth_session=<JWT>...` + `{ success: true }` |
| 全エラーレスポンス | 429/401/400/500 + 本文: `"コードが正しくありません"` |

### 4.3 POST /api/chat

**ファイル**: `src/app/api/chat/route.ts`

```typescript
export const runtime = 'nodejs'  // jose Node API + インメモリ rate-limiter 使用

export async function POST(request: NextRequest): Promise<Response> {
  // 1. 多層防御: proxy.ts とは独立した JWT 再検証
  const cookieStore = await cookies()
  const token = cookieStore.get(AUTH_COOKIE_NAME)?.value
  if (!token || !(await verifyAuthCookie(token))) {
    return new Response('Unauthorized', { status: 401 })
  }

  // 2. リクエスト解析
  let body: {
    id?: string
    messages?: UIMessage[]
    modelId?: string
    effort?: EffortId
    thinking?: boolean
  }
  try {
    body = await request.json()
  } catch {
    return new Response('Bad Request', { status: 400 })
  }

  const { messages, modelId, effort, thinking } = body
  if (!Array.isArray(messages) || messages.length === 0) {
    return new Response('Bad Request', { status: 400 })
  }

  // 3. モデル ID 検証（許可リスト。不正値はデフォルトへフォールバック）
  const resolvedModel = ALLOWED_MODEL_IDS.includes(modelId ?? '')
    ? (modelId as string)
    : DEFAULT_MODEL_ID

  // 4. Effort レベル解決（不正値はデフォルトへフォールバック）
  const effortLevel =
    EFFORT_LEVELS.find(e => e.id === effort) ??
    EFFORT_LEVELS.find(e => e.id === DEFAULT_EFFORT_ID)!

  // 5. Thinking 適用判定（thinking フラグ && モデルの supportsThinking）
  const modelMeta = MODELS.find(m => m.id === resolvedModel)
  const applyThinking = thinking === true && modelMeta?.supportsThinking === true

  // 6. ストリーミング実行
  // thinking と temperature は排他: thinking 有効時は temperature を設定しない
  const result = streamText({
    model: anthropic(resolvedModel),
    system: SYSTEM_PROMPT,
    messages: await convertToModelMessages(messages),
    ...(applyThinking
      ? {
          providerOptions: {
            anthropic: {
              thinking: { type: 'enabled', budgetTokens: effortLevel.budgetTokens }
            }
          }
        }
      : { temperature: effortLevel.temperature }
    )
  })

  // 7. AI SDK UIMessageStream 形式で返却
  return result.toUIMessageStreamResponse()
}
```

**ANTHROPIC_API_KEY**: コード内に記述なし。`anthropic()` プロバイダが `process.env.ANTHROPIC_API_KEY` を暗黙参照する。

## 10. ミドルウェア詳細設計（proxy.ts）

```typescript
// Edge ランタイム
// ファイル: src/proxy.ts（named export 'proxy'）
// ※ Next.js 慣習の middleware.ts / default export ではない

// 定数インライン（Edge では constants.ts を import 不可のため）
const AUTH_COOKIE_NAME = 'auth_session'
const PUBLIC_PATHS = ['/auth', '/api/auth/verify', '/api/auth/logout']

function isPublicPath(pathname: string): boolean {
  return PUBLIC_PATHS.some(p => pathname === p || pathname.startsWith(p + '/'))
}

function getSecret(): Uint8Array {
  const secret = process.env.COOKIE_SECRET
  if (!secret) throw new Error('COOKIE_SECRET not configured')
  // ※ cookies.ts と異なり 32文字チェックは行わない
  return new TextEncoder().encode(secret)
}

export async function proxy(request: NextRequest): Promise<NextResponse> {
  const { pathname } = request.nextUrl

  if (isPublicPath(pathname)) return NextResponse.next()

  const token = request.cookies.get(AUTH_COOKIE_NAME)?.value
  if (!token) return NextResponse.redirect(new URL('/auth', request.url))

  try {
    await jwtVerify(token, getSecret(), { algorithms: ['HS256'] })
    return NextResponse.next()
  } catch {
    const res = NextResponse.redirect(new URL('/auth', request.url))
    res.cookies.delete(AUTH_COOKIE_NAME)
    return res
  }
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)']
}
```
